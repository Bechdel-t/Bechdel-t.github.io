const tripsFindByPrice = async (req, res) => {
  const { min, max } = req.query;
  try {
    const trips = await Model.find({
      perPerson: { $gte: Number(min), $lte: Number(max) }
    }).exec();
    return res.status(200).json(trips);
  } catch (err) {
    return res.status(500).json(err);
  }
};
